function(compile_shader name profile entry)
    set(source "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${name}.hlsl")
    set(output "${SHADER_OUTPUT_DIR}/${name}.spv")
    set(extra_source)
    if(name MATCHES "_bindless")
        string(REPLACE "_bindless" "" base_name "${name}")
        set(extra_source "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${base_name}.hlsl")
    endif()
    if(name MATCHES "^shadow_terrain_stream")
        list(APPEND extra_source "${CMAKE_CURRENT_SOURCE_DIR}/shaders/terrain_stream.vert.hlsl")
    endif()
    if(name STREQUAL "shadow_terrain.vert")
        list(APPEND extra_source "${CMAKE_CURRENT_SOURCE_DIR}/shaders/terrain.vert.hlsl")
    endif()
    if(name STREQUAL "visibility_frustum.comp")
        set(extra_source "${CMAKE_CURRENT_SOURCE_DIR}/shaders/visibility.comp.hlsl")
    endif()
    if(name STREQUAL "terrain_height_fixture.comp")
        set(extra_source "${CMAKE_CURRENT_SOURCE_DIR}/shaders/terrain_tile_height.comp.hlsl")
    endif()
    if(name STREQUAL "terrain_normal_fixture.comp")
        set(extra_source "${CMAKE_CURRENT_SOURCE_DIR}/shaders/terrain_tile_normal.comp.hlsl")
    endif()
    if(name MATCHES "_temporal")
        string(REPLACE "_temporal" "" temporal_base "${name}")
        string(REPLACE "_bindless" "" temporal_base "${temporal_base}")
        set(extra_source "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${temporal_base}.hlsl")
    endif()
    set(shader_environment vulkan1.3)
    if(name MATCHES "terrain_stream.*frag" OR name STREQUAL "terrain_depth.frag" OR name MATCHES "mesh.*frag" OR name MATCHES "shadow_.*frag")
        # Core OpKill is sufficient for the view-boundary clip. Targeting SPIR-V
        # 1.5 avoids requiring the optional shaderDemoteToHelperInvocation feature.
        set(shader_environment vulkan1.2)
    endif()
    set(shader_options)
    if(name MATCHES "\.frag$")
        # Match the shared vertex interface without requiring maintenance4.
        list(APPEND shader_options -fspv-preserve-interface)
    endif()
    add_custom_command(OUTPUT "${output}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${SHADER_OUTPUT_DIR}"
        COMMAND "${DXC_EXECUTABLE}" -spirv -fspv-target-env=${shader_environment}
            -T "${profile}" -E "${entry}" -Ges -WX -O3 ${shader_options}
            -Fo "${output}" "${source}"
        DEPENDS "${source}" ${extra_source} ${SHADER_INCLUDES}
        COMMENT "Compiling HLSL: ${name} (${profile})"
        VERBATIM)
    set(SHADER_BINARIES ${SHADER_BINARIES} "${output}" PARENT_SCOPE)
endfunction()
